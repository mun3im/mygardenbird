# MyGardenBird pipeline dependencies.
#
# Not every stage needs everything here. The groups below say which stages use
# what, so you can install a subset: building the dataset needs neither
# TensorFlow nor the GUI toolkit, and training needs neither the GUI nor the
# Xeno-canto client.

# ---- core: used throughout -------------------------------------------------
numpy>=1.26
pandas>=2.0
tqdm>=4.65

# ---- audio I/O and analysis (Stages 4-7, 9) --------------------------------
librosa>=0.10
soundfile>=0.12            # Stage 5: FLAC/WAV read and write
scipy>=1.11

# ---- Stage 1-2: Xeno-canto retrieval ---------------------------------------
requests>=2.31

# ---- Stage 4: annotation GUIs ----------------------------------------------
# Two annotators share stage4_core.py and produce identical label files.
# Stage4_annotate_qt.py (the one in routine use) needs all three packages
# below; Stage4_annotate_segments.py, the original matplotlib tool, needs only
# matplotlib and sounddevice.
PySide6>=6.5               # Qt bindings
pyqtgraph>=0.13            # fast spectrogram/waveform rendering
sounddevice>=0.4           # audio playback; needs PortAudio on the system

# ---- Stage 8: MIP splitting ------------------------------------------------
PuLP>=2.8                  # bundles the CBC solver

# ---- Stage 7: BirdNET validation -------------------------------------------
# Note these are two different version numbers. birdnet_analyzer is the Python
# PACKAGE; the MODEL it ships is BirdNET GLOBAL 6K V2.4, which Stage 7 pins by
# filename, so the model is fixed whichever of these package versions you have.
# The published results were produced with package 2.4.0 and model V2.4.
# The upper bound is deliberate: a future major release could ship a different
# model or move the checkpoint paths, which would silently change the results.
# The package is MIT licensed; the model weights are released under a
# non-commercial licence (see NOTICE).
birdnet_analyzer>=2.2,<3

# ---- Stage 9: training (optional unless you are training) ------------------
tensorflow>=2.15
scikit-learn>=1.3
psutil>=5.9                # resource monitoring during training

# ---- plotting --------------------------------------------------------------
matplotlib>=3.8
seaborn>=0.13
